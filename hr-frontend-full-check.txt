    83	export function getApiError(
    84	  error,
    85	  fallback = "The request could not be completed."
    86	) {
    87	  if (!error) {
    88	    return fallback;
    89	  }
    90	
    91	  if (typeof error === "string") {
    92	    return error;
    93	  }
    94	
    95	  if (error.message) {
    96	    return error.message;
    97	  }
    98	
    99	  const data = error.response?.data;
   100	
   101	  if (typeof data === "string") {
   102	    return data;
   103	  }
   104	
   105	  if (data?.detail) {
   106	    return data.detail;
   107	  }
   108	
   109	  if (data?.message) {
   110	    return data.message;
   111	  }
   112	
   113	  if (data?.error) {
   114	    return data.error;
   115	  }
   116	
   117	  if (data && typeof data === "object") {
   118	    const firstField = Object.keys(data)[0];
   119	    const firstError = data[firstField];
   120	
   121	    if (Array.isArray(firstError)) {
   122	      return `${firstField}: ${firstError[0]}`;
   123	    }
   124	
   125	    if (typeof firstError === "string") {
   126	      return `${firstField}: ${firstError}`;
   127	    }
   128	  }
   129	
   130	  return fallback;
   131	}
